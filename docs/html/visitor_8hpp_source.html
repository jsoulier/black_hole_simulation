67 template<SavepointMemcpyable T, typename... Args>
68 void operator()(T& item, SavepointVersion version = {}, Args&&... args)
69 {
70 // For detecting bugs in MSVC concepts
71 static_assert(!SavepointRange<T>);
72 static_assert(!SavepointPointer<T>);
73 if (IsReading())
74 {
75 // Required for write-only containers (e.g. views)
76 if constexpr (std::is_const_v<T>)
77 {
78 SavepointLog("Tried to read into a const");
79 return;
80 }
81 else
82 {
83 if (Version < version)
84 {
85 if constexpr (sizeof...(Args) > 0)
86 {
87 item = T{std::forward<Args>(args)...};
88 }
89 return;
90 }
91 if (sizeof(T) > GetSize())
92 {
93 SavepointLog(std::format("Tried to read past visitor: {} -> {}", Version.GetString(), version.GetString()));
94 if constexpr (sizeof...(Args) > 0)
95 {
96 item = T{std::forward<Args>(args)...};
97 }
98 return;
99 }
100 std::memcpy(std::addressof(item), Reader.data() + Offset, sizeof(T));
101 Offset += sizeof(T);
102 }
103 }
104 else
105 {
106 Writer.resize(Writer.size() + sizeof(T));
107 std::memcpy(Writer.data() + Writer.size() - sizeof(T), std::addressof(item), sizeof(T));
108 }
109 }

169 template<SavepointMemcpyable T>
170 void Skip(size_t size = 1)
171 {
172 if (IsWriting())
173 {
174 SavepointLog("Tried to skip on a writer");
175 return;
176 }
177 if (!size)
178 {
179 SavepointLog("Tried to skip nothing");
180 return;
181 }
182 if (sizeof(T) * size > GetSize())
183 {
184 SavepointLog(std::format("Tried to skip past visitor: {}", Version.GetString()));

186 }

282template<SavepointStdPointer T>
283void SavepointVisit(SavepointVisitor& visitor, T& item)
284{
285 using E = typename T::element_type;
286 if (visitor.IsReading())
287 {
288 bool hasPointer = false;
289 visitor(hasPointer);
290 if (!hasPointer)
291 {
292 return;
293 }
294 if (!item)
295 {
296 if constexpr (std::is_base_of_v<SavepointBase, E>)
297 {
298 item.reset(dynamic_cast<E*>(SavepointReadDerived(visitor)));
299 return;
300 }
301 else if constexpr (std::is_default_constructible_v<E>)
302 {
303 item.reset(new E());
304 if (!item)
305 {
306 SavepointLog("Failed to allocate pointer");
307 visitor.Fail();
308 return;
309 }
310 }
311 else
312 {
313 // Don't static_assert because it'll fail on already instanciated
314 // derived classes with abstract parents
315 SavepointLog("No method to create pointer");
316 }
317 }
318 visitor(*item);
319 }
320 else
321 {
322 bool hasPointer = item.get() != nullptr;
323 visitor(hasPointer);
324 if (hasPointer)
325 {
326 if constexpr (std::is_base_of_v<SavepointBase, E>)
327 {
328 SavepointWriteDerived(item.get(), visitor);
329 }
330 else
331 {
332 visitor(*item);
333 }
334 }
335 }
336}

362template<SavepointRange T>
363void SavepointVisit(SavepointVisitor& visitor, T& item)
364{
365 using E = typename T::value_type;
366 size_t size = item.size();
367 if constexpr (SavepointDynamicRange<T>)
368 {
369 if (visitor.IsReading() && size)
370 {
371 SavepointLog("Tried to read into non-empty range");
372 item.clear();
373 }
374 }
375 visitor(size);
376 if (visitor.IsReading())
377 {
378 // Can detect when we read garbage and would iterate forever
379 if (size > visitor.GetSize())
380 {
381 SavepointLog("Tried to read past visitor");
382 visitor.Fail();
383 return;
384 }
385 if constexpr (SavepointDynamicRange<T>)
386 {
387 auto inserter = std::inserter(item, std::ranges::end(item));
388 for (size_t i = 0; i < size; i++)
389 {
390 // TODO: mutable iterators
391 E element;
392 visitor(element);
393 *inserter++ = element;
394 }
395 }
396 else if constexpr (SavepointStaticRange<T>)
397 {
398 size_t maxSize = std::ranges::size(item);
399 if (size > maxSize)
400 {
401 SavepointLog(std::format("Fixed range is too small: {} < {}", maxSize, size));
402 }
403 if (size < maxSize)
404 {
405 SavepointLog(std::format("Fixed range will be truncated: {} < {}", maxSize, size));
406 }
407 maxSize = std::min(size, maxSize);
408 for (size_t i = 0; i < maxSize; i++)
409 {
410 visitor(item[i]);
411 }
412 for (; maxSize < size; maxSize++)
413 {
414 visitor.Skip<E>();
415 }
416 }
417 else
418 {
419 // Required for write-only containers (e.g. views)
420 SavepointLog("Unknown range");
421 }
422 }
423 else
424 {
425 for (auto& element : item)
426 {
427 visitor(element);
428 }
429 }
